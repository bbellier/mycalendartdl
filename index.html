// Rappel quotidien par mail des tâches étoilées non terminées dont l'échéance tombe dans 7 jours ou demain. // Un mail par compte, envoyé à l'adresse de connexion de chaque utilisateur. import { createClient } from "jsr:@supabase/supabase-js@2"; Deno.serve(async (req) => { // Seul le cron (porteur du secret) peut déclencher la fonction if (req.headers.get("x-cron-secret") !== Deno.env.get("CRON_SECRET")) return new Response("Non autorisé", { status: 401 }); const sb = createClient(Deno.env.get("SUPABASE_URL")!, Deno.env.get("SUPABASE_SERVICE_ROLE_KEY")!); // Dates cibles (J+1 et J+7) à l'heure de Paris const jour = (n: number) => new Intl.DateTimeFormat("en-CA", { timeZone: "Europe/Paris" }).format(new Date(Date.now() + n * 864e5)); const j1 = jour(1), j7 = jour(7); // Tâches étoilées concernées, tous comptes confondus (catégorie facultative) const { data, error } = await sb.from("taches") .select("user_id, titre, echeance, notes, categories(nom, icone, couleur)") .eq("etoile", true).eq("fait", false).in("echeance", [j1, j7]).order("echeance"); if (error) return new Response(error.message, { status: 500 }); if (!data.length) return new Response("Aucune tâche à rappeler"); // Mise en forme aux couleurs DATAfutée : une section « Demain », une section « Dans 7 jours » const esc = (s: string) => s.replace(/[&<>"]/g, (c) => ({ "&": "&", "<": "<", ">": ">", '"': """ }[c]!)); const fmt = (s: string) => new Date(s + "T12:00").toLocaleDateString("fr-FR", { weekday: "long", day: "numeric", month: "long" }); const ligne = (t: any) => ` ${esc(t.categories?.icone ?? "⭐")} ${esc(t.titre)}
${t.categories ? esc(t.categories.nom) + " · " : ""}${fmt(t.echeance)} ${t.notes ? `
${esc(t.notes)}` : ""}`; const bloc = (titre: string, ts: any[]) => ts.length ? `

${titre}

${ts.map(ligne).join("")}
` : ""; // Regroupement par compte, puis un mail par compte const parUser: Record = {}; data.forEach((t: any) => (parUser[t.user_id] ??= []).push(t)); const bilan: string[] = []; for (const [uid, ts] of Object.entries(parUser)) { const { data: u, error: eu } = await sb.auth.admin.getUserById(uid); if (eu || !u.user?.email) { bilan.push(`${uid} : adresse introuvable`); continue; } const demain = ts.filter((t) => t.echeance === j1), semaine = ts.filter((t) => t.echeance === j7); const html = `
⭐ Rappel de l'agenda
${bloc("Demain", demain)}${bloc("Dans 7 jours", semaine)}
`; const sujet = demain.length ? `Rappel : ${demain.length} échéance(s) demain` + (semaine.length ? `, ${semaine.length} dans 7 jours` : "") : `Rappel : ${semaine.length} échéance(s) dans 7 jours`; // Envoi via Resend (sans domaine vérifié, Resend n'accepte que l'adresse du compte Resend) const r = await fetch("https://api.resend.com/emails", { method: "POST", headers: { Authorization: `Bearer ${Deno.env.get("RESEND_API_KEY")}`, "Content-Type": "application/json" }, body: JSON.stringify({ from: Deno.env.get("MAIL_FROM") ?? "Agenda ", to: [u.user.email], subject: sujet, html }), }); bilan.push(`${u.user.email} : ${r.status}`); } return new Response(bilan.join("\n")); });
